-- 1. Lista todos os clientes com seus respectivos treinos
SELECT c.nome AS Cliente, t.nome_treino AS Treino
FROM cliente c
INNER JOIN cliente_treino ct ON c.id_cliente = ct.id_cliente
INNER JOIN treino t ON ct.id_treino = t.id_treino;

-- 2. Conta quantos clientes estão com status "Pendente"
SELECT status, COUNT(*) AS Total
FROM cliente_treino
GROUP BY status;

-- 3. Mostra todos os treinos e seus instrutores responsáveis
SELECT t.nome_treino, i.nome_instrutor
FROM treino t
JOIN instrutor i ON t.id_instrutor = i.id_instrutor;

-- 4. Mostra exercícios do treino de maior dificuldade
SELECT t.nome_treino, e.nome_exercicio
FROM treino t
JOIN treino_exercicio te ON t.id_treino = te.id_treino
JOIN exercicio e ON te.id_exercicio = e.id_exercicio
WHERE t.nivel_dificuldade = 'Avançado';

-- 5. Mostra a média de frequência semanal por nível de dificuldade
SELECT nivel_dificuldade, AVG(frequencia_semanal) AS media_frequencia
FROM treino
GROUP BY nivel_dificuldade;
